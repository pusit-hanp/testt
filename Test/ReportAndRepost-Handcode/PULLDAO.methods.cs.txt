private string ReportRepostHeaderSql()
{
    // WIPHEADER is used only when the slip does not exist in TPHEADER.
    // A blank TPHEADER.SAP_DOC1 must never fall back to WIPHEADER.
    return @"SELECT MISLIP_NO, MAX(SAP_DOC1) SAP_DOC1,
                    MAX(ISSUE_BY) ISSUE_BY, COUNT(*) HEADER_COUNT
             FROM (
                 SELECT MISLIP_NO, SAP_DOC1, ISSUE_BY FROM TPHEADER
                 UNION ALL
                 SELECT w.MISLIP_NO, w.SAP_DOC1, w.ISSUE_BY FROM WIPHEADER w
                 WHERE NOT EXISTS (
                     SELECT 1 FROM TPHEADER t WHERE t.MISLIP_NO = w.MISLIP_NO
                 )
             ) GROUP BY MISLIP_NO";
}

public ReportAndRepostPage GetReportAndRepost(ReportAndRepostCriteria criteria, int draw, int start, int length)
{
    DateTime from = DateTime.ParseExact(criteria.DATE_FROM, "MM/dd/yyyy", CultureInfo.InvariantCulture);
    DateTime to = DateTime.ParseExact(criteria.DATE_TO, "MM/dd/yyyy", CultureInfo.InvariantCulture);
    if (from > to) throw new ArgumentException("Date From must not exceed Date To.");
    if (start < 0 || length < 1 || length > 25) throw new ArgumentException("Invalid page.");

    DynamicParameters param = new DynamicParameters();
    param.Add("DATE_FROM", from);
    param.Add("DATE_TO_EXCLUSIVE", to.AddDays(1));
    StringBuilder where = new StringBuilder(" WHERE fifo.APPROVE_DATE >= :DATE_FROM AND fifo.APPROVE_DATE < :DATE_TO_EXCLUSIVE ");
    string[] columns = { "CTN", "PN", "BATCH", "MI_SLIPNUMBER" };
    string[] values = { criteria.CTN, criteria.PN, criteria.BATCH, criteria.MI_SLIPNUMBER };
    for (int i = 0; i < columns.Length; i++)
    {
        if (String.IsNullOrWhiteSpace(values[i])) continue;
        where.Append(" AND fifo." + columns[i] + " LIKE :" + columns[i]);
        param.Add(columns[i], values[i].Trim());
    }
    if (!String.IsNullOrWhiteSpace(criteria.FROM_LOCATION) && criteria.FROM_LOCATION != "-1")
    {
        where.Append(" AND fifo.FROM_LOCATION = :FROM_LOCATION");
        param.Add("FROM_LOCATION", criteria.FROM_LOCATION.Trim());
    }
    if (!String.IsNullOrWhiteSpace(criteria.TO_LOCATION) && criteria.TO_LOCATION != "-1")
    {
        where.Append(" AND fifo.TO_LOCATION = :TO_LOCATION");
        param.Add("TO_LOCATION", criteria.TO_LOCATION.Trim());
    }

    using (var session = new SessionFactory(base.ConnectionString))
    {
        var statuses = session.Exec<ReportAndRepostStatusCount>(
            "SELECT fifo.STATUS_TRANFER, COUNT(*) TOTAL FROM PULLLIST_FIFOLOG fifo" + where +
            " GROUP BY fifo.STATUS_TRANFER", param).ToList();
        string mode = String.IsNullOrEmpty(criteria.STATUS_MODE) ? "FAIL" : criteria.STATUS_MODE;
        if (!new[] { "ALL", "FAIL", "SUCCESS", "EMPTY", "UNKNOWN", "EXACT" }.Contains(mode))
            throw new ArgumentException("Invalid status filter.");
        var selected = statuses.Where(x => mode == "ALL" ||
            (mode == "EXACT" ? String.Equals(x.STATUS_TRANFER, criteria.STATUS_VALUE, StringComparison.Ordinal) :
             ReportAndRepostStatus.GetKind(x.STATUS_TRANFER) == mode)).ToList();
        var page = new ReportAndRepostPage {
            draw = draw, recordsTotal = statuses.Sum(x => x.TOTAL),
            recordsFiltered = selected.Sum(x => x.TOTAL),
            statuses = statuses, data = new List<ReportAndRepostRow>()
        };
        if (page.recordsFiltered == 0) return page;

        if (mode != "ALL")
        {
            List<string> statusConditions = new List<string>();
            for (int i = 0; i < selected.Count; i++)
            {
                if (selected[i].STATUS_TRANFER == null) statusConditions.Add("fifo.STATUS_TRANFER IS NULL");
                else
                {
                    string key = "STATUS" + i;
                    statusConditions.Add("fifo.STATUS_TRANFER = :" + key);
                    param.Add(key, selected[i].STATUS_TRANFER);
                }
            }
            where.Append(" AND (" + String.Join(" OR ", statusConditions) + ")");
        }
        param.Add("PAGE_START", start);
        param.Add("PAGE_END", checked(start + length));
        string sql = @"SELECT p.*, h.SAP_DOC1, h.ISSUE_BY, NVL(h.HEADER_COUNT, 0) HEADER_COUNT,
                    (SELECT COUNT(*) FROM PULLLIST_FIFOLOG d
                     WHERE d.CTN = p.CTN AND d.MI_SLIPNUMBER = p.MI_SLIPNUMBER) TRANSACTION_ROWS
                FROM (
                    SELECT numbered.* FROM (
                        SELECT ROWIDTOCHAR(fifo.ROWID) ROW_ID,
                            fifo.MI_SLIPNUMBER, fifo.CTN, fifo.PN, fifo.BATCH, fifo.QTY,
                            fifo.FROM_LOCATION, fifo.TO_LOCATION, fifo.REASON,
                            fifo.APPROVE_DATE, fifo.ACT_BIN, fifo.RFID_TAG, fifo.STATUS_TRANFER,
                            ROW_NUMBER() OVER (ORDER BY fifo.APPROVE_DATE DESC, fifo.ROWID) RN
                        FROM PULLLIST_FIFOLOG fifo " + where + @"
                    ) numbered WHERE numbered.RN > :PAGE_START AND numbered.RN <= :PAGE_END
                ) p
                LEFT JOIN (" + ReportRepostHeaderSql() + @") h ON h.MISLIP_NO = p.MI_SLIPNUMBER
                ORDER BY p.RN";
        page.data = session.Exec<ReportAndRepostRow>(sql, param).ToList();
        foreach (var row in page.data)
        {
            row.BLOCK_REASON = ReportAndRepostStatus.GetBlockReason(row);
            row.CAN_REPOST = row.BLOCK_REASON == "";
        }
        return page;
    }
}

public ReportAndRepostResult RepostReportCtn(string rowId,
    Func<ReportAndRepostRow, ReportAndRepostSmartReply> sendToSmart)
{
    var result = new ReportAndRepostResult { ROW_ID = rowId, Outcome = "SKIPPED" };
    using (var session = new SessionFactory(base.ConnectionString))
    {
        if (session.Conn.State != ConnectionState.Open) session.Conn.Open();
        using (var transaction = session.Conn.BeginTransaction())
        {
            bool smartSucceeded = false;
            bool smartUncertain = false;
            try
            {
                DynamicParameters param = new DynamicParameters();
                param.Add("ROW_ID", rowId);
                var row = session.Conn.Query<ReportAndRepostRow>(@"
                    SELECT ROWIDTOCHAR(fifo.ROWID) ROW_ID,
                        fifo.MI_SLIPNUMBER, fifo.CTN, fifo.PN, fifo.BATCH, fifo.STATUS_TRANFER
                    FROM PULLLIST_FIFOLOG fifo
                    WHERE fifo.ROWID = CHARTOROWID(:ROW_ID)
                    FOR UPDATE NOWAIT", param, transaction).SingleOrDefault();
                if (row == null)
                {
                    result.Message = "Row no longer exists. Search again.";
                    transaction.Rollback();
                    return result;
                }
                result.CTN = row.CTN;
                result.MI_SLIPNUMBER = row.MI_SLIPNUMBER;
                DynamicParameters headerParam = new DynamicParameters();
                headerParam.Add("SLIP", row.MI_SLIPNUMBER);
                var header = session.Conn.Query<ReportAndRepostRow>(
                    "SELECT * FROM (" + ReportRepostHeaderSql() + ") WHERE MISLIP_NO = :SLIP",
                    headerParam, transaction).SingleOrDefault();
                if (header != null)
                {
                    row.SAP_DOC1 = header.SAP_DOC1;
                    row.ISSUE_BY = header.ISSUE_BY;
                    row.HEADER_COUNT = header.HEADER_COUNT;
                }
                DynamicParameters identity = new DynamicParameters();
                identity.Add("CTN", row.CTN);
                identity.Add("SLIP", row.MI_SLIPNUMBER);
                row.TRANSACTION_ROWS = session.Conn.Query<int>(@"
                    SELECT COUNT(*) FROM PULLLIST_FIFOLOG
                    WHERE CTN = :CTN AND MI_SLIPNUMBER = :SLIP", identity, transaction).Single();
                string reason = ReportAndRepostStatus.GetBlockReason(row);
                if (reason != "")
                {
                    result.Message = reason;
                    transaction.Rollback();
                    return result;
                }

                // Keep the DB row locked while the existing HTTP call style runs in Controller.
                // The callback avoids adding RestSharp to the DAL or creating an interface.
                ReportAndRepostSmartReply reply = sendToSmart(row);
                smartUncertain = reply.IsUncertain;
                smartSucceeded = reply.Success && !smartUncertain;
                if (!smartSucceeded)
                {
                    result.Outcome = reply.IsUncertain ? "SMART_RESULT_UNKNOWN" : "FAILED";
                    result.Message = reply.IsUncertain
                        ? "SmartWH result is unknown. Check before sending again. " + reply.Detail
                        : reply.Detail;
                    transaction.Rollback();
                    return result;
                }

                param.Add("NEW_STATUS", ReportAndRepostStatus.SuccessStatus);
                param.Add("OLD_STATUS", row.STATUS_TRANFER);
                param.Add("SLIP", row.MI_SLIPNUMBER);
                param.Add("CTN", row.CTN);
                int changed = session.Conn.Execute(@"
                    UPDATE PULLLIST_FIFOLOG SET STATUS_TRANFER = :NEW_STATUS
                    WHERE ROWID = CHARTOROWID(:ROW_ID)
                      AND MI_SLIPNUMBER = :SLIP AND CTN = :CTN
                      AND STATUS_TRANFER = :OLD_STATUS", param, transaction);
                if (changed != 1) throw new InvalidOperationException("Expected to update exactly one log row.");
                transaction.Commit();
                result.Success = true;
                result.Outcome = "SUCCESS";
                result.Message = "SmartWH confirmed success; Pull List status updated.";
            }
            catch (Exception ex)
            {
                try { transaction.Rollback(); } catch { }
                if (smartUncertain)
                {
                    // Cleanup failure must not make an unknown SmartWH result safe to retry.
                    result.Outcome = "SMART_RESULT_UNKNOWN";
                    result.Message += " DB cleanup failed: " + ex.Message;
                }
                else
                {
                    result.Outcome = smartSucceeded ? "SMART_SUCCESS_DB_FAILED" : "FAILED";
                    result.Message = smartSucceeded
                        ? "SmartWH confirmed success, but DB save was not confirmed. Check before sending again. " + ex.Message
                        : ex.Message;
                }
            }
        }
    }
    return result;
}
